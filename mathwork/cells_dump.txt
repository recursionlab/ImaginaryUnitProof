4: class WallFieldSimulation:
5:     def __init__(self, num_nodes):
6:         self.num_nodes = num_nodes
7:         # Initialize V (amplitude) to 0.0 and phi (phase) to 0.0 for all nodes
8:         self.V = np.zeros(num_nodes)
9:         self.phi = np.zeros(num_nodes)
10:         # Adjacency matrix for LISTEN weights
11:         self.weights = np.zeros((num_nodes, num_nodes))
12: 
13:     def set_node_state(self, node_idx, V_val, phi_val):
14:         self.V[node_idx] = V_val
15:         self.phi[node_idx] = phi_val
16: 
17:     def set_listen_weight(self, source, target, weight):
18:         self.weights[source, target] = weight
19: 
20:     def compute_listen_laplacian(self):
21:         """
22:         Computes the Laplacian matrix of the LISTEN graph.
23:         L = D - W, where D is the diagonal out-degree matrix.
24:         """
25:         # Out-degrees
26:         D = np.diag(np.sum(self.weights, axis=1))
27:         L = D - self.weights
28:         return L
29: 
30:     def get_algebraic_connectivity(self):
31:         """
32:         Computes lambda_2 (algebraic connectivity) of the symmetric part of the Laplacian.
33:         """
34:         # For directed graphs, we typically look at the symmetric laplacian of the underlying undirected graph
35:         sym_weights = (self.weights + self.weights.T) / 2.0
36:         D = np.diag(np.sum(sym_weights, axis=1))
37:         L_sym = D - sym_weights
38: 
39:         eigenvalues = np.sort(la.eigvalsh(L_sym))
40:         # lambda_2 is the second smallest eigenvalue
41:         if len(eigenvalues) > 1:
42:             return eigenvalues[1]
43:         return eigenvalues[0]

5: class RefactoredWallFieldSimulation:
6:     def __init__(self, num_nodes):
7:         self.num_nodes = num_nodes
8:         self.V = np.zeros(num_nodes)
9:         self.phi = np.zeros(num_nodes)
10:         self.weights = np.zeros((num_nodes, num_nodes))
11: 
12:     def set_node_state(self, node_idx, V_val, phi_val):
13:         self.V[node_idx] = V_val
14:         self.phi[node_idx] = phi_val
15: 
16:     def set_listen_weight(self, source, target, weight):
17:         self.weights[source, target] = weight
18: 
19:     def compute_listen_laplacian(self):
20:         D = np.diag(np.sum(self.weights, axis=1))
21:         return D - self.weights
22: 
23:     def get_algebraic_connectivity(self):
24:         sym_weights = (self.weights + self.weights.T) / 2.0
25:         D = np.diag(np.sum(sym_weights, axis=1))
26:         L_sym = D - sym_weights
27:         eigenvalues = np.sort(la.eigvalsh(L_sym))
28:         return eigenvalues[1] if len(eigenvalues) > 1 else eigenvalues[0]
29: 
30:     def get_phase_alignment(self, node_a, node_b):
31:         """
32:         Calculates the phase difference (alignment) between two nodes.
33:         """
34:         return self.phi[node_a] - self.phi[node_b]
35: 
36:     def tickVField(self, alpha=0.1):
37:         """
38:         Simulates one time-step of signal diffusion.
39:         The transmission of V from source to target is gated by the phase alignment:
40:         Effective Weight = weight * cos(delta_phi)
41:         """
42:         new_V = np.copy(self.V)
43:         for i in range(self.num_nodes):
44:             diffusion_sum = 0.0
45:             for j in range(self.num_nodes):
46:                 if self.weights[j, i] > 0:
47:                     d_phi = self.phi[j] - self.phi[i]
48:                     # Gating factor: cos(d_phi) models how phase alignment affects signal flow
49:                     gating = np.cos(d_phi)
50:                     diffusion_sum += self.weights[j, i] * gating * self.V[j]
51: 
52:             # Apply diffusion update
53:             new_V[i] = self.V[i] + alpha * (diffusion_sum - np.sum(self.weights[i, :]) * self.V[i])
54: 
55:         self.V = np.clip(new_V, 0.0, 10.0) # Prevent divergence
